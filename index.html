// Простой сайт, где есть кнопка "Пробить телефон". // При нажатии на кнопку выводится инфа, собранная по IP (для примера показываем IP и user-agent). // HTML разметка document.body.innerHTML = `

Пробить телефон

По IP определим предполагаемые данные о посетителе.

`; async function lookupByIP() { const resultDiv = document.getElementById('result'); resultDiv.innerHTML = 'Поиск...'; try { // Получаем IP const response = await fetch('https://api.ipify.org?format=json'); const ip_data = await response.json(); const ip = ip_data.ip; // Получаем geo-инфу (можно использовать ipapi.co или ipinfo.io) const geo = await fetch(`https://ipapi.co/${ip}/json/`).then(res => res.json()); // Выводим собранную инфу resultDiv.innerHTML = ` IP: ${ip}
Страна: ${geo.country_name || '-'}
Город: ${geo.city || '-'}
Регион: ${geo.region || '-'}
Оператор (Org): ${geo.org || '-'}
Ваш браузер: ${navigator.userAgent}

* Телефон по IP в бытовых условиях узнать невозможно, но вот пример информации, которую можно получить по вашему подключению. `; } catch (e) { resultDiv.innerHTML = 'Ошибка получения данных :('; } } document.getElementById('lookupBtn').onclick = lookupByIP;
